(function () {
  var EMNER = ['Vlog','Infosec','Product','Dev','Marketing','It','Crypto','Fintech','Design','Web3','DevOps','AI','Data','Tech'];
  var ANDET = 'More articles';
  var alle = [], grupper = {}, raekke = [];
  function esc(s) { return String(s).replace(/[&<>"']/g, function (c) { return ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]); }); }
  function dato(s) { return s ? s.slice(0, 10) : ''; }
  function hent(start) { return fetch('/feeds/posts/summary?alt=json&start-index=' + start + '&max-results=150').then(function (r) { return r.json(); }); }
  function byg() {
    EMNER.slice().reverse().forEach(function (e) { grupper[e] = []; });
    grupper[ANDET] = [];
    alle.forEach(function (p) {
      var valgt = null;
      for (var i = 0; i < EMNER.length; i++) { if (p.labels.indexOf(EMNER[i]) > -1) { valgt = EMNER[i]; break; } }
      grupper[valgt || ANDET].push(p);
    });
    raekke = Object.keys(grupper).filter(function (k) { return grupper[k].length > 0; });
    raekke.sort(function (a, b) { if (a === ANDET) return 1; if (b === ANDET) return -1; return grupper[b].length - grupper[a].length; });
    var ud = raekke.map(function (navn, nr) {
      var poster = grupper[navn].slice().sort(function (a, b) { return b.dato.localeCompare(a.dato); });
      return '<div class="tma-idx-group' + (nr === 0 ? ' open' : '') + '"><button class="tma-idx-head" type="button"><svg class="caret" width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3"><path d="M9 5l7 7-7 7"></path></svg>' + esc(navn) + '<span class="n">' + poster.length + '</span></button><ul class="tma-idx-list">' + poster.map(function (p) { return '<li><a href="' + esc(p.url) + '"><span>' + esc(p.titel) + '</span><time>' + esc(p.dato) + '</time></a></li>'; }).join('') + '</ul></div>';
    }).join('');
    document.getElementById('tmaIdxGroups').innerHTML = ud;
    document.getElementById('tmaIdxCount').textContent = alle.length + ' articles';
  }
  function sider() {
    fetch('/feeds/pages/default?alt=json&max-results=50').then(function (r) { return r.json(); }).then(function (j) {
      var e = (j.feed.entry || []).map(function (x) { var l = (x.link || []).filter(function (y) { return y.rel === 'alternate'; })[0] || {}; return { titel: x.title.$t, url: l.href || '#' }; })
        .filter(function (x) { return !/^Nothing here|^Sitemap$/i.test(x.titel); })
        .sort(function (a, b) { return a.titel.localeCompare(b.titel); });
      document.getElementById('tmaIdxPages').innerHTML = e.map(function (x) { return '<li><a href="' + esc(x.url) + '">' + esc(x.titel) + '</a></li>'; }).join('');
    }).catch(function () { document.getElementById('tmaIdxPages').innerHTML = ''; });
  }
  function filtrer() {
    var q = document.getElementById('tmaIdxSearch').value.trim().toLowerCase(), vist = 0;
    document.querySelectorAll('.tma-idx-group').forEach(function (g) {
      var n = 0;
      g.querySelectorAll('li').forEach(function (li) { var m = !q || li.textContent.toLowerCase().indexOf(q) > -1; li.style.display = m ? '' : 'none'; if (m) n++; });
      g.querySelector('.n').textContent = n; g.style.display = n ? '' : 'none';
      if (q && n) g.classList.add('open');
      vist += n;
    });
    document.getElementById('tmaIdxCount').textContent = q ? vist + ' of ' + alle.length + ' articles' : alle.length + ' articles';
  }
  document.addEventListener('click', function (ev) {
    var h = ev.target.closest && ev.target.closest('.tma-idx-head');
    if (h) h.parentNode.classList.toggle('open');
    if (ev.target.id === 'tmaIdxToggleAll') {
      var luk = ev.target.textContent === 'Collapse all';
      document.querySelectorAll('.tma-idx-group').forEach(function (g) { g.classList.toggle('open', !luk); });
      ev.target.textContent = luk ? 'Expand all' : 'Collapse all';
    }
  });
  document.addEventListener('input', function (ev) { if (ev.target.id === 'tmaIdxSearch') filtrer(); });
  Promise.all([hent(1), hent(130)]).then(function (svar) {
    var set = {};
    svar.forEach(function (j) { (j.feed.entry || []).forEach(function (e) {
      var l = (e.link || []).filter(function (y) { return y.rel === 'alternate'; })[0] || {};
      if (!l.href || set[l.href]) return; set[l.href] = 1;
      alle.push({ titel: e.title.$t, url: l.href, dato: dato((e.published || {}).$t), labels: (e.category || []).map(function (c) { return c.term; }) });
    }); });
    byg(); sider();
  }).catch(function () {
    document.getElementById('tmaIdxGroups').innerHTML = '<p class="tma-idx-empty">The article list could not be loaded. You can browse everything in <a href="/sitemap.xml">sitemap.xml</a>.</p>';
    document.getElementById('tmaIdxCount').textContent = '';
  });
})();
